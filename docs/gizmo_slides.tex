% Slide 2: The Core Problem
\begin{frame}{The Core Problem: Cardinality-Constrained Portfolio Selection}
\begin{columns}
\begin{column}{0.55\textwidth}
\textbf{Markowitz Mean-Variance Formulation (1952):}
\begin{equation*}
\min_{\mathbf{x} \in \{0, 1\}^N} \mathcal{C}(\mathbf{x}) = \frac{q}{2} \mathbf{x}^T \boldsymbol{\Sigma} \mathbf{x} - \boldsymbol{\mu}^T \mathbf{x}
\end{equation*}
subject to the exact budget/cardinality constraint:
\begin{equation*}
\sum_{i=0}^{N-1} x_i = K \quad \text{(Pick exactly } K \text{ out of } N \text{ assets)}
\end{equation*}
\begin{itemize}
    \item $\boldsymbol{\mu} \in \mathbb{R}^N$: Asset expected annual returns.
    \item $\boldsymbol{\Sigma} \in \mathbb{R}^{N \times N}$: Return covariance matrix.
    \item $q > 0$: Risk-aversion coefficient.
    \item \textbf{Complexity:} NP-hard binary quadratic optimization.
\end{itemize}
\end{column}

\begin{column}{0.43\textwidth}
\begin{block}{The Combinatorial Reality}
\begin{itemize}
    \item Total search space: $2^N$ states.
    \item Feasible asset combinations: $\binom{N}{K}$.
    \item \textbf{Example ($N=16, K=4$):}
    \begin{itemize}
        \item Total states: $65,536$
        \item Feasible states: $1,820$
        \item \alert{Valid fraction: only $2.78\%$}
    \end{itemize}
    \item How does standard gate-model quantum optimization handle this constraint?
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% Slide 3: Why Conventional QAOA Fails
\begin{frame}{The Conventional Approach \& Its Catastrophic Failure}
\textbf{The Standard Textbook Recipe:} Map the constraint to a soft penalty term:
\begin{equation*}
H_{\text{total}} = H_C + \lambda \left(\sum_{i=0}^{N-1} x_i - K\right)^2, \quad H_M = \sum_{i=0}^{N-1} X_i, \quad |\psi_0\rangle = |+\rangle^{\otimes N}
\end{equation*}

\vspace{0.3cm}
\begin{columns}
\begin{column}{0.48\textwidth}
\begin{alertblock}{Failure 1: The Penalty Parameter Dilemma}
\begin{itemize}
    \item \textbf{$\lambda$ too small:} Quantum state collapses into unconstrained space; $>60\%$ of measured shots violate the budget ($\sum x_i \neq K$).
    \item \textbf{$\lambda$ too large:} Penalty canyons dominate the cost landscape; optimizer ignores variance $\boldsymbol{\Sigma}$ and returns $\boldsymbol{\mu}$.
\end{itemize}
\end{alertblock}
\end{column}

\begin{column}{0.48\textwidth}
\begin{alertblock}{Failure 2: Exponential State Dilution}
\begin{itemize}
    \item Transverse mixer $\sum X_i$ breaks Hamming weight conservation.
    \item Amplitudes leak indiscriminately into the $2^N$ unconstrained space.
    \item On noisy NISQ hardware, $>95\%$ of experimental shot budget is wasted on illegal portfolios.
\end{itemize}
\end{alertblock}
\end{column}
\end{columns}
\end{frame}

% Slide 4: The Gizmo Architecture
\begin{frame}{The Gizmo Architecture: Complete Subspace Confinement}
\begin{center}
\textbf{Core Thesis: Stop penalizing invalid configurations. Confine the Hilbert space.}
\end{center}

\vspace{0.2cm}
\begin{columns}
\begin{column}{0.48\textwidth}
\begin{block}{1. Deterministic Dicke Initialization}
\begin{itemize}
    \item Initialize in the symmetric superposition:
    \begin{equation*}
    |D_K^N\rangle = \binom{N}{K}^{-1/2} \sum_{\text{wt}(\mathbf{x})=K} |\mathbf{x}\rangle
    \end{equation*}
    \item Synthesized via an inductive $\mathcal{O}(NK)$ controlled-Givens rotation ladder (Bärtschi \& Eidenbenz).
    \item Zero classical statevector allocation.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{2. Excitation-Preserving $XY$-Mixer}
\begin{itemize}
    \item Replace $X_i$ with hopping interactions:
    \begin{equation*}
    H_{XY} = \frac{1}{2} \sum_{i=0}^{N-2} \left(X_i X_{i+1} + Y_i Y_{i+1}\right)
    \end{equation*}
    \item Commutes with total excitation:
    \begin{equation*}
    \left[H_{XY}, \sum_{i=0}^{N-1} Z_i\right] = 0
    \end{equation*}
    \item \textbf{Guarantees 100\% feasible measurement samples.}
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% Slide 5: The Math
\begin{frame}{Mathematical Rigor: Commutation Proof \& Subspace Invariance}
\textbf{Analytical Commutator Derivation on Edge $(i, j)$:}
\begin{align*}
[X_i X_j + Y_i Y_j, Z_i + Z_j] &= [X_i X_j, Z_i] + [X_i X_j, Z_j] + [Y_i Y_j, Z_i] + [Y_i Y_j, Z_j] \\
&= (-2i Y_i X_j) + (-2i X_i Y_j) + (2i X_i Y_j) + (2i Y_i X_j) \\
&= 0
\end{align*}

\begin{columns}
\begin{column}{0.5\textwidth}
\begin{block}{Direct Consequence}
For any evolution angle $\beta \in \mathbb{R}$:
\begin{equation*}
\left[e^{-i \beta H_{XY}}, \sum_{i=0}^{N-1} Z_i\right] = 0
\end{equation*}
The unitary $U_{XY}(\beta)$ leaves the eigenspaces of total Hamming weight strictly invariant.
\end{block}
\end{column}

\begin{column}{0.48\textwidth}
\begin{block}{Elimination of Hyperparameters}
\begin{itemize}
    \item Penalty multiplier $\lambda$ is completely eliminated.
    \item Classical outer optimizer (COBYLA) searches strictly within the $\binom{N}{K}$ financial subspace.
    \item Every measurement shot is a valid allocation.
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% Slide 6: Heavy-Hex Compilation & SWAP Network
\begin{frame}{Heavy-Hex Compilation: Linear $\mathcal{O}(N)$ Alternating SWAP Network}
\begin{itemize}
    \item \textbf{The Bottleneck:} Return covariance $\boldsymbol{\Sigma}$ is dense $\implies \binom{N}{2}$ all-to-all $Z_i Z_j$ interactions.
    \item Mapping $K_N$ onto IBM heavy-hex (degree $\le 3$) via standard heuristic transpilation creates dozens of uncontrolled SWAPs.
\end{itemize}

\vspace{0.2cm}
\begin{columns}
\begin{column}{0.55\textwidth}
\begin{block}{Linear Alternating SWAP Network}
\begin{itemize}
    \item Arrange $N$ qubits on a 1D nearest-neighbor chain.
    \item In each stage: apply $R_{ZZ}(2\gamma J_{q_a, q_b})$ on adjacent physical wires, followed by SWAP.
    \item Alternating even/odd physical pairs executes all $\frac{N(N-1)}{2}$ interactions in \textbf{exactly $N$ stages ($\mathcal{O}(N)$ depth)}.
\end{itemize}
\end{block}
\end{column}

\begin{column}{0.43\textwidth}
\begin{block}{Permutation Unrolling}
\begin{itemize}
    \item After $N$ stages, physical wire $p$ holds logical asset $\sigma(p)$.
    \item \textbf{The Innovation:} Measure physical wire $p$ directly into classical register bit $\sigma(p)$.
    \item \textbf{Zero inverse-SWAP gate overhead!}
\end{itemize}
\end{block}
\end{column}
\end{columns}
\end{frame}

% Slide 9: Classical Skepticism & The HUBO Frontier
\begin{frame}{Epistemic Honesty: Why Not Just Use Classical MIP / Gurobi?}
\begin{alertblock}{Intellectual Honesty Check}
For standard quadratic binary programs with $N \le 50$, classical mixed-integer solvers (Gurobi, CPLEX, SCIP) reach global optimality in milliseconds. We make no false claims of beating classical MIPs on toy quadratic models.
\end{alertblock}

\vspace{0.2cm}
\begin{block}{Where Quantum Advantage Actually Resides: Higher-Order Binary Optimization (HUBO)}
\begin{itemize}
    \item \textbf{Nonlinear Market Impact:} Piecewise cubic transaction costs ($c_i x_i^{3/2}$) where classical convex relaxations loosen.
    \item \textbf{Higher-Order Tail Risk:} Asymmetric downside risk incorporating co-skewness ($S_{ijk} x_i x_j x_k$) and co-kurtosis ($K_{ijkl} x_i x_j x_k x_l$) risk tensors.
    \item \textbf{The Frontier:} Classical branch-and-cut faces worst-case exponential tree exploration under HUBO, whereas gate-model quantum superposition naturally executes multi-qubit phase rotations.
\end{itemize}
\end{block}
